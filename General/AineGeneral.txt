local function resolve(text)
	local query = text:match("^%s*(.-)%s*$"):gsub("^@", ""):lower()
	if query == "" then return nil, "Enter a player name" end
	local displayMatches, prefixMatches = {}, {}
	for _, candidate in ipairs(Players:GetPlayers()) do
		if candidate ~= player then
			local username, display = candidate.Name:lower(), candidate.DisplayName:lower()
			if username == query then return candidate end
			if display == query then table.insert(displayMatches, candidate) end
			if username:sub(1, #query) == query or display:sub(1, #query) == query then
				table.insert(prefixMatches, candidate)
			end
		end
	end
	local matches = #displayMatches > 0 and displayMatches or prefixMatches
	if #matches == 1 then return matches[1] end
	return nil, #matches > 1 and "Multiple matches; use @username" or "Player not found"
end
local function characterParts(who)
	local character = who.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if humanoid and root and humanoid.Health > 0 then return character, humanoid, root end
end
local function modelAimPart(model)
	local humanoid = model and model:FindFirstChildOfClass("Humanoid")
	local root = model and (model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart)
	if humanoid and root and humanoid.Health > 0 then
		return model:FindFirstChild("Head") or root, humanoid, model
	end
end
local function aimPart(target)
	if not target then return end
	if target.player then
		local character, humanoid, root = characterParts(target.player)
		if character then return character:FindFirstChild("Head") or root, humanoid, character end
	elseif target.model and target.model:IsDescendantOf(workspace) then
		return modelAimPart(target.model)
	end
end

local function setAimTarget(target, lockNow)
	aimTarget = target
	aimLocked = target ~= nil and lockNow == true
	nextAimReacquire = 0
	if target then
		if target.player then aimInput.Text = "@" .. target.player.Name else aimInput.Text = "" end
		aimStatus.Text = (aimLocked and "Locked: " or "Selected: ") .. targetLabel(target)
	else
		aimStatus.Text = "No target selected"
	end
	paintAim()
	return target ~= nil
end
local function visibleFromCamera(camera, model, part)
	local character = player.Character
	local parameters = RaycastParams.new()
	parameters.FilterType = Enum.RaycastFilterType.Exclude
	parameters.FilterDescendantsInstances = character and {character} or {}
	parameters.IgnoreWater = true
	local origin = camera.CFrame.Position
	local hit = workspace:Raycast(origin, part.Position - origin, parameters)
	return not hit or hit.Instance:IsDescendantOf(model)
end
local function targetInView()
	local camera = workspace.CurrentCamera
	if not camera then return nil, "Camera unavailable" end
	local center = camera.ViewportSize / 2
	local radius = math.max(90, math.min(camera.ViewportSize.X, camera.ViewportSize.Y) * 0.22)
	local best, bestDistance
	for _, model in ipairs(workspace:GetDescendants()) do
		if model:IsA("Model") and model ~= player.Character then
			local part, _, character = modelAimPart(model)
			if part then
				local owner = Players:GetPlayerFromCharacter(character)
				local point, onScreen = camera:WorldToViewportPoint(part.Position)
				if onScreen and point.Z > 0 then
					local distance = (Vector2.new(point.X, point.Y) - center).Magnitude
					if distance <= radius and (not bestDistance or distance < bestDistance)
						and visibleFromCamera(camera, character, part) then
						bestDistance = distance
						best = owner and {player = owner, name = owner.Name}
							or {model = character, name = character.Name}
					end
				end
			end
		end
	end
	return best, best and nil or "Put a visible target near screen center"
end
local function chooseAimFromInput(lockNow)
	local target, message
	if aimInput.Text:match("^%s*$") then
		target, message = targetInView()
	else
		local who
		who, message = resolve(aimInput.Text)
		if who then target = {player = who, name = who.Name} end
	end
	if not target then aimStatus.Text = message; return end
	setAimTarget(target, lockNow)
end

local function reacquireAimModel()
	if not aimTarget or aimTarget.player then return end
	local best, bestDistance
	for _, candidate in ipairs(workspace:GetDescendants()) do
		if candidate:IsA("Model") and candidate.Name == aimTarget.name then
			local part = modelAimPart(candidate)
			if part then
				local distance = aimTarget.lastPosition and (part.Position - aimTarget.lastPosition).Magnitude or 0
				if not bestDistance or distance < bestDistance then
					best, bestDistance = candidate, distance
				end
			end
		end
	end
	if best then
		aimTarget.model = best
		aimStatus.Text = "Reacquired: " .. aimTarget.name
	end
end
local function updateAim(dt)
	aimFrameTime = aimFrameTime * 0.9 + math.min(dt, 0.1) * 0.1
	if not aimLocked or not aimTarget then return end
	local camera = workspace.CurrentCamera
	if not camera then return end
	local part = aimPart(aimTarget)
	if not part then
		if not aimTarget.player and os.clock() >= nextAimReacquire then
			nextAimReacquire = os.clock() + 0.5
			reacquireAimModel()
			part = aimPart(aimTarget)
		end
		if not part then
			aimStatus.Text = "Waiting for " .. targetLabel(aimTarget) .. " to return"
			return
		end
	end
	aimTarget.lastPosition = part.Position
	if os.clock() >= nextAimPing then
		nextAimPing = os.clock() + 1
		pcall(function()
			local item = Stats.Network.ServerStatsItem["Data Ping"]
			local milliseconds = item and item:GetValue()
			if type(milliseconds) == "number" then aimPing = math.clamp(milliseconds / 1000, 0, 0.6) end
		end)
		aimStatus.Text = string.format("Locked: %s  %dms / %dfps",
			targetLabel(aimTarget), math.floor(aimPing * 1000 + 0.5),
			math.floor(1 / math.max(aimFrameTime, 1 / 240) + 0.5))
	end
	local velocity = part.AssemblyLinearVelocity
	if velocity.Magnitude > 180 then velocity = velocity.Unit * 180 end
	local leadTime = math.clamp(aimPing * 0.5 + aimFrameTime * 1.5, 0.02, 0.35)
	local predicted = part.Position + velocity * leadTime
	camera.CFrame = CFrame.lookAt(camera.CFrame.Position, predicted, camera.CFrame.UpVector)
end
RunService:BindToRenderStep("AineGeneralAim", Enum.RenderPriority.Camera.Value + 1, function(dt)
	if closed then return end
	local ok, message = pcall(updateAim, dt)
	if not ok then
		aimLocked = false
		aimStatus.Text = "Aim error: " .. tostring(message)
		paintAim()
	end
end)
local function stopMotion(root)
	if root and root.Parent then
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	end
end
local function stopFlight()
	local active = flight
	if not active then return end
	flight = nil
	active.velocity:Destroy()
	active.gyro:Destroy()
	if active.humanoid.Parent then
		active.humanoid.PlatformStand = active.platformStand
		active.humanoid.AutoRotate = active.autoRotate
		if player.Character == active.character and active.humanoid.Health > 0 and not active.platformStand then
			active.humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
		end
	end
	stopMotion(active.root)
end
local function updateFlight(character, humanoid, root)
	local camera = workspace.CurrentCamera
	if not camera then stopFlight(); return end
	if flight and flight.root ~= root then stopFlight() end
	if not flight then
		flight = {
			character = character, humanoid = humanoid, root = root,
			platformStand = humanoid.PlatformStand, autoRotate = humanoid.AutoRotate,
			velocity = make("BodyVelocity", {
				Name = "GeneralFlyVelocity", P = 10000, Velocity = Vector3.zero,
				MaxForce = Vector3.new(math.huge, math.huge, math.huge),
			}, root),
			gyro = make("BodyGyro", {
				Name = "GeneralFlyGyro", P = 100000, CFrame = root.CFrame,
				MaxTorque = Vector3.new(math.huge, math.huge, math.huge),
			}, root),
		}
	end
	humanoid.PlatformStand = true
	humanoid.AutoRotate = false
	local movement = UserInputService:GetFocusedTextBox() and Vector3.zero or humanoid.MoveDirection
	local look = camera.CFrame.LookVector
	local forward = Vector3.new(look.X, 0, look.Z)
	if forward.Magnitude < 0.001 then
		local facing = root.CFrame.LookVector
		forward = Vector3.new(facing.X, 0, facing.Z)
	end
	if forward.Magnitude < 0.001 then forward = Vector3.new(0, 0, -1) end
	forward = forward.Unit
	local amount = movement:Dot(forward)
	local direction = movement - forward * amount + look * amount
	if direction.Magnitude > 1 then direction = direction.Unit end
	flight.velocity.Velocity = direction * 50
	flight.gyro.CFrame = camera.CFrame
end
local function restoreCollisions()
	for part, saved in pairs(collisionState) do
		if part.Parent then part.CanCollide = saved end
	end
	table.clear(collisionState)
end
local function finishBurst()
	local active = burst
	if not active then return end
	burst = nil
	if active.force then active.force:Destroy() end
	if active.spin then active.spin:Destroy() end
	if active.thrust then active.thrust:Destroy() end
	if active.humanoid.Parent then
		active.humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, active.seated)
		active.humanoid.AutoRotate = active.autoRotate
	end
	if active.physical then
		for part, state in pairs(active.physical) do
			if part.Parent then
				part.CustomPhysicalProperties = state.properties
				part.Massless = state.massless
			end
		end
	end
	if active.camera and active.camera.Parent and active.cameraSubject then
		active.camera.CameraSubject = active.cameraSubject
	end
	if player.Character == active.character and active.root.Parent then
		active.character:PivotTo(active.origin * CFrame.new(0, 0.5, 0))
		stopMotion(active.root)
		if active.humanoid.Health > 0 then active.humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end
	end
	nextBurst = os.clock() + 0.35
end
local function setRow(row, enabled)
	if row == flingRow then finishBurst() end
	row.enabled = false
	row.target = nil
	if enabled then
		local target, message = resolve(row.input.Text)
		row.target = target
		row.enabled = target ~= nil
		row.status.Text = target and (target.DisplayName .. "(@" .. target.Name .. ")") or message
	else
		row.status.Text = "Stopped"
	end
	paintToggle(row.toggle, row.enabled)
end

	end)
end

local function saveTeleport()
	local _, _, root = characterParts(player)
	if not root or burst then teleportFeedback("WAIT"); return end
	savedTP = root.CFrame
	teleportButton.BackgroundColor3 = Color3.fromRGB(0, 180, 95)
	teleportFeedback("SAVED")
end
local function teleportSaved()
	if not savedTP then teleportFeedback("SET FIRST"); return end
	local _, _, root = characterParts(player)
	if not root then teleportFeedback("WAIT"); return end
	if tpRow.enabled then setRow(tpRow, false) end
	if flingRow.enabled then setRow(flingRow, false) end
	finishBurst()
	stopFlight()
	root.CFrame = savedTP
	stopMotion(root)
	teleportFeedback("TP")
end
local function trackTeleportGesture(position)
	local gesture = tpGesture
	if not gesture then return end
	gesture.last = position
	local dx, dy = position.X - gesture.start.X, position.Y - gesture.start.Y
	gesture.travel = math.max(gesture.travel, math.sqrt(dx * dx + dy * dy))
end
connect(teleportButton.InputBegan, function(input)
	if tpGesture then return end
	if input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
	tpGesture = {input = input, start = input.Position, last = input.Position, travel = 0}
end)
connect(UserInputService.InputChanged, function(input)
	if not tpGesture then return end
	if input == tpGesture.input or (tpGesture.input.UserInputType == Enum.UserInputType.MouseButton1
		and input.UserInputType == Enum.UserInputType.MouseMovement) then
		trackTeleportGesture(input.Position)
	end
end)
connect(UserInputService.InputEnded, function(input)
	local gesture = tpGesture
	if not gesture or input ~= gesture.input then return end
	trackTeleportGesture(input.Position)
	tpGesture = nil
	if input.UserInputState == Enum.UserInputState.Cancel then return end
	local dx, dy = gesture.last.X - gesture.start.X, gesture.last.Y - gesture.start.Y
	if dy <= -24 and -dy > math.abs(dx) then
		saveTeleport()
	elseif gesture.travel <= 12 then
		teleportSaved()
	end
end)
connect(teleportButton.Activated, function(input)
	if not input or (input.UserInputType ~= Enum.UserInputType.Touch
		and input.UserInputType ~= Enum.UserInputType.MouseButton1) then teleportSaved() end
end)

local function removeESP(who)
	local objects = espObjects[who]
	if objects then
		objects.billboard:Destroy()
		for _, edge in ipairs(objects.edges) do edge:Destroy() end
		espObjects[who] = nil
	end
end
local function clearESP()
	for who in pairs(espObjects) do removeESP(who) end
end
quickToggle("ESP", 0, function(update)
	esp = not esp
	if not esp then clearESP() end
	espElapsed = 0.1
	update(esp)
end, true)
local function updateESP()
	for _, who in ipairs(Players:GetPlayers()) do
		if who ~= player then
			local character, _, root = characterParts(who)
			local objects = espObjects[who]
			if not character then
				removeESP(who)
			else
				if objects and objects.character ~= character then removeESP(who); objects = nil end
				if not objects then
					local billboard = make("BillboardGui", {
						Name = "PlayerNameESP", Adornee = root, AlwaysOnTop = true,
						Size = UDim2.fromOffset(280, 28),
					}, gui)
					local name = label(billboard, "", UDim2.fromOffset(0, 0), UDim2.fromScale(1, 1), 14)
					name.TextXAlignment = Enum.TextXAlignment.Center
					name.TextStrokeTransparency = 0.2
					name.TextStrokeColor3 = Color3.new(0, 0, 0)
					local edges = {}
					for index = 1, 12 do
						edges[index] = make("BoxHandleAdornment", {
							Name = "PlayerBoxEdge", Adornee = root, AlwaysOnTop = true,
							ZIndex = 1, Color3 = colors.accent, Transparency = 0.1,
						}, gui)
					end
					objects = {character = character, billboard = billboard, name = name, edges = edges}
					espObjects[who] = objects
				end
				local bounds, size = character:GetBoundingBox()
				local relative = root.CFrame:ToObjectSpace(bounds)
				local edgeIndex = 0
				for _, a in ipairs({-1, 1}) do
					for _, b in ipairs({-1, 1}) do
						for axis = 1, 3 do
							edgeIndex = edgeIndex + 1
							local edge = objects.edges[edgeIndex]
							local offset = axis == 1 and Vector3.new(0, a * size.Y / 2, b * size.Z / 2)
								or axis == 2 and Vector3.new(a * size.X / 2, 0, b * size.Z / 2)
								or Vector3.new(a * size.X / 2, b * size.Y / 2, 0)
							edge.Adornee = root
							edge.Size = axis == 1 and Vector3.new(size.X, 0.035, 0.035)
								or axis == 2 and Vector3.new(0.035, size.Y, 0.035)
								or Vector3.new(0.035, 0.035, size.Z)
							edge.CFrame = relative * CFrame.new(offset)
						end
					end
				end
				objects.billboard.Adornee = root
				objects.billboard.StudsOffsetWorldSpace = Vector3.new(0, bounds.Position.Y - root.Position.Y + size.Y / 2 + 0.7, 0)
				objects.name.Text = who.DisplayName .. "(@" .. who.Name .. ")"
			end
		end
	end
end

local function flingPart(character, root)
	local head = character and character:FindFirstChild("Head")
	if root and head and (root.Position - head.Position).Magnitude <= 5 then return root end
	if head then return head end
	if root then return root end
	local accessory = character and character:FindFirstChildOfClass("Accessory")
	return accessory and accessory:FindFirstChild("Handle")
end

local function startBurst(target, character, humanoid, root)
	stopFlight()
	restoreCollisions()
	local targetCharacter = target.Character
	local targetHumanoid = targetCharacter and targetCharacter:FindFirstChildOfClass("Humanoid")
	local targetRoot = targetCharacter and (targetHumanoid and targetHumanoid.RootPart
		or targetCharacter:FindFirstChild("HumanoidRootPart"))
	local targetPart = flingPart(targetCharacter, targetRoot)
	burst = {
		target = target, targetCharacter = targetCharacter, targetPart = targetPart,
		character = character, humanoid = humanoid, root = root, origin = root.CFrame,
		started = os.clock(), phase = 0, nextStep = 0,
		camera = workspace.CurrentCamera,
		cameraSubject = workspace.CurrentCamera and workspace.CurrentCamera.CameraSubject,
		physical = {},
		seated = humanoid:GetStateEnabled(Enum.HumanoidStateType.Seated), autoRotate = humanoid.AutoRotate,
	}
	for _, part in ipairs(character:GetDescendants()) do
		if part:IsA("BasePart") then
			burst.physical[part] = {
				properties = part.CustomPhysicalProperties,
				massless = part.Massless,
			}
			part.CustomPhysicalProperties = PhysicalProperties.new(100, 0.3, 0.5)
			part.Massless = true
			part.Velocity = Vector3.zero
			part.RotVelocity = Vector3.zero
		end
	end
	humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
	humanoid.Sit = false
	humanoid.AutoRotate = false
	burst.spin = make("BodyAngularVelocity", {
		Name = "GeneralFlingSpin", AngularVelocity = Vector3.new(0, 99999, 0),
		MaxTorque = Vector3.new(0, math.huge, 0), P = math.huge,
	}, root)
	burst.thrust = make("BodyThrust", {
		Name = "GeneralFlingThrust", Force = Vector3.new(9e9, 0, 9e9),
		Location = root.Position,
	}, root)
	if burst.camera then
		burst.camera.CameraSubject = targetPart or targetHumanoid or humanoid
	end
end
local function moveBurstCharacter(character, root, pivot)
	root.CFrame = pivot
	if character.PrimaryPart then
		character:SetPrimaryPartCFrame(pivot)
	else
		character:PivotTo(pivot)
	end
end
local function physicsStep()
	local character, humanoid, root = characterParts(player)
	if burst then
		local targetCharacter, targetHumanoid, targetRoot = characterParts(burst.target)
		local targetPart = flingPart(targetCharacter, targetRoot)
		if not flingRow.enabled or not root or root ~= burst.root or not targetRoot or not targetPart
			or burst.target.Parent ~= Players or burst.target.Character ~= burst.targetCharacter
			or targetHumanoid.Sit or targetPart.Velocity.Magnitude > 500
			or os.clock() - burst.started >= 2 then
			finishBurst()
		elseif os.clock() >= burst.nextStep then
			burst.nextStep = os.clock() + 0.03
			burst.phase = burst.phase + 1
			local phase = burst.phase
			if phase % 4 == 1 then
				-- The far pulse is the part used by the classic invis-fling
				-- routines to build momentum before restoring contact position.
				moveBurstCharacter(character, root, root.CFrame + Vector3.new(0, 9999, 0))
				root.RotVelocity = Vector3.new(0, 99999, 0)
				root.AssemblyAngularVelocity = Vector3.new(0, 99999, 0)
				return
			end
			local moving = targetPart.Velocity.Magnitude >= 50
			local pivot
			if not moving then
				local lead = targetHumanoid.MoveDirection * targetPart.Velocity.Magnitude / 1.25
				local positions = {
					{Vector3.new(0, 1.5, 0), CFrame.Angles(math.rad(phase * 100), 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(math.rad(phase * 100), 0, 0)},
					{Vector3.new(2.25, 1.5, -2.25), CFrame.Angles(math.rad(phase * 100), 0, 0)},
					{Vector3.new(-2.25, -1.5, 2.25), CFrame.Angles(math.rad(phase * 100), 0, 0)},
					{Vector3.new(0, 1.5, 0), CFrame.Angles(math.rad(phase * 100), 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(math.rad(phase * 100), 0, 0)},
				}
				local position = positions[(phase - 1) % #positions + 1]
				pivot = CFrame.new(targetPart.Position + position[1] + lead) * position[2]
			else
				local speed = targetPart.Velocity.Magnitude / 1.25
				local positions = {
					{Vector3.new(0, 1.5, targetHumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0)},
					{Vector3.new(0, -1.5, -targetHumanoid.WalkSpeed), CFrame.Angles(0, 0, 0)},
					{Vector3.new(0, 1.5, targetHumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0)},
					{Vector3.new(0, 1.5, speed), CFrame.Angles(math.rad(90), 0, 0)},
					{Vector3.new(0, -1.5, -speed), CFrame.Angles(0, 0, 0)},
					{Vector3.new(0, 1.5, speed), CFrame.Angles(math.rad(90), 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(0, 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0)},
					{Vector3.new(0, -1.5, 0), CFrame.Angles(0, 0, 0)},
				}
				local position = positions[(phase - 1) % #positions + 1]
				pivot = CFrame.new(targetPart.Position + position[1]) * position[2]
			end
			moveBurstCharacter(character, root, pivot)
			root.RotVelocity = Vector3.new(0, 99999, 0)
			root.AssemblyAngularVelocity = Vector3.new(0, 99999, 0)
		end
	end
	if not root then stopFlight(); return end
	if burst then return end
	if flingRow.enabled and os.clock() >= nextBurst and not UserInputService:GetFocusedTextBox() then
		local _, targetHumanoid, targetRoot = characterParts(flingRow.target)
		if targetRoot and not targetHumanoid.Sit then
			startBurst(flingRow.target, character, humanoid, root)
			return
		end
	end
	if noclip then
		for _, part in ipairs(character:GetDescendants()) do
			if part:IsA("BasePart") then
				if collisionState[part] == nil then collisionState[part] = part.CanCollide end
				part.CanCollide = false
			end
		end
	end
	if antiFling then
		if root.AssemblyLinearVelocity.Magnitude > 500 then root.AssemblyLinearVelocity = Vector3.zero end
		if root.AssemblyAngularVelocity.Magnitude > 500 then root.AssemblyAngularVelocity = Vector3.zero end
	end
	if fly and not tpRow.enabled then updateFlight(character, humanoid, root) else stopFlight() end
	if tpRow.enabled and not UserInputService:GetFocusedTextBox() then
		local _, _, targetRoot = characterParts(tpRow.target)
		if targetRoot then
			root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
			stopMotion(root)
		end
	end
end

local function cleanup()
	if closed then return end
	closed = true
	tpGesture = nil
	RunService:UnbindFromRenderStep("AineGeneralAim")
	for _, connection in ipairs(connections) do connection:Disconnect() end
	finishBurst()
	stopFlight()
	restoreCollisions()
	clearESP()
	gui:Destroy()
end
shutdown.OnInvoke = cleanup
connect(close.Activated, cleanup)
connect(gui.Destroying, cleanup)

end)

connect(RunService.Stepped, function()
	if closed then return end
	local ok, message = pcall(physicsStep)
	if not ok then
		finishBurst()
		setRow(flingRow, false)
		setRow(tpRow, false)
		warn("General controls: " .. tostring(message))
	end
end)
connect(RunService.Heartbeat, function(dt)
	if closed then return end
	if tpFlashUntil > 0 and os.clock() >= tpFlashUntil then
		teleportButton.Text = savedTP and "TP" or "SET TP"
		tpFlashUntil = 0
	end
	if not esp then return end
	espElapsed = espElapsed + dt
	if espElapsed < 0.1 then return end
	espElapsed = 0
	updateESP()
end)
